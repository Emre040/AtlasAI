## Reproducible plotting code

This code produces: (1) the log-scale scatter plot, (2) a diverging bar chart for the 15 most discordant proteins, and (3) the secretome-location count bar chart.

```python
import duckdb
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# `con` should be a DuckDB connection containing the HPA `proteinatlas` table.
# For example: con = duckdb.connect("hpa.duckdb")

sql = """
WITH d AS (
    SELECT
        Gene AS gene,
        "Secretome location" AS secretome_location,
        TRY_CAST("Blood concentration - Conc. blood IM [pg/L]" AS DOUBLE) AS im_pg_l,
        TRY_CAST("Blood concentration - Conc. blood MS [pg/L]" AS DOUBLE) AS ms_pg_l
    FROM proteinatlas
)
SELECT
    gene,
    secretome_location,
    im_pg_l,
    ms_pg_l,
    LOG10(im_pg_l) AS log10_im,
    LOG10(ms_pg_l) AS log10_ms,
    LOG10(im_pg_l / ms_pg_l) AS log10_im_over_ms
FROM d
WHERE im_pg_l > 0 AND ms_pg_l > 0
"""

d = con.execute(sql).df()
d["secretome_location"] = d["secretome_location"].fillna("(not annotated)")
d["abs_log10_ratio"] = d["log10_im_over_ms"].abs()

# 1. Scatter: immunoassay against mass spectrometry
plt.figure(figsize=(8, 6))
sns.scatterplot(
    data=d,
    x="log10_ms",
    y="log10_im",
    hue="secretome_location",
    s=45,
    alpha=0.8
)
lims = [
    min(d["log10_ms"].min(), d["log10_im"].min()),
    max(d["log10_ms"].max(), d["log10_im"].max())
]
plt.plot(lims, lims, "--", color="black", linewidth=1, label="IM = MS")
plt.xlim(lims)
plt.ylim(lims)
plt.xlabel("log10 mass-spectrometry concentration (pg/L)")
plt.ylabel("log10 immunoassay concentration (pg/L)")
plt.title("Blood concentrations: immunoassay versus mass spectrometry")
plt.tight_layout()
plt.show()

# 2. Diverging chart: 15 greatest absolute log10 ratios
top15 = (
    d.sort_values(["abs_log10_ratio", "gene"], ascending=[False, True])
     .head(15)
     .sort_values("log10_im_over_ms")
)

plt.figure(figsize=(9, 7))
colors = top15["log10_im_over_ms"].map(
    lambda x: "#c44e52" if x < 0 else "#4c72b0"
)
plt.barh(top15["gene"], top15["log10_im_over_ms"], color=colors)
plt.axvline(0, color="black", linewidth=0.8)
plt.xlabel("log10(immunoassay / mass spectrometry)")
plt.ylabel("Gene")
plt.title("Fifteen most discordant blood concentrations")
plt.tight_layout()
plt.show()

# 3. Number of proteins per secretome location
location_counts = (
    d.groupby("secretome_location", dropna=False)
     .size()
     .reset_index(name="protein_count")
     .sort_values("protein_count", ascending=False)
)

plt.figure(figsize=(7, 4))
sns.barplot(
    data=location_counts,
    x="secretome_location",
    y="protein_count",
    color="#55a868"
)
plt.xlabel("Secretome location")
plt.ylabel("Number of proteins")
plt.title("Proteins with both blood-concentration measurements")
plt.tight_layout()
plt.show()

print(location_counts)
print(top15[
    ["gene", "secretome_location", "im_pg_l", "ms_pg_l",
     "log10_im", "log10_ms", "log10_im_over_ms"]
])
```
