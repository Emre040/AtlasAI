## Plotting code

The following code generates the requested scatter plot and the two diverging bar charts from the HPA table extracted by the query.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# df should contain one row per included protein with these columns:
# Gene, im, ms, grp
#
# grp values:
# "Liver-enriched" and "Rest"

df["log10_im"] = np.log10(df["im"])
df["log10_ms"] = np.log10(df["ms"])
df["log10_ratio"] = df["log10_im"] - df["log10_ms"]

colors = {
    "Liver-enriched": "#D55E00",
    "Rest": "#0072B2"
}

sns.set_theme(style="whitegrid")

# 1. Scatter plot
fig, ax = plt.subplots(figsize=(8, 6))

for group in ["Liver-enriched", "Rest"]:
    sub = df[df["grp"] == group]
    ax.scatter(
        sub["log10_ms"],
        sub["log10_im"],
        s=35,
        alpha=0.75,
        label=f"{group} (n={len(sub)})",
        color=colors[group],
        edgecolor="none"
    )

lims = [
    min(df["log10_ms"].min(), df["log10_im"].min()),
    max(df["log10_ms"].max(), df["log10_im"].max())
]
ax.plot(lims, lims, "--", color="gray", linewidth=1)

ax.set_xlabel("log10 mass-spectrometry concentration (pg/L)")
ax.set_ylabel("log10 immunoassay concentration (pg/L)")
ax.set_title("Blood concentration: immunoassay versus mass spectrometry")
ax.legend()
plt.tight_layout()
plt.show()


# 2. Diverging bar charts for the ten largest positive ratios
fig, axes = plt.subplots(1, 2, figsize=(14, 7), sharex=False)

for ax, group in zip(axes, ["Liver-enriched", "Rest"]):
    top = (
        df[df["grp"] == group]
        .nlargest(10, "log10_ratio")
        .sort_values("log10_ratio")
    )

    ax.barh(
        top["Gene"],
        top["log10_ratio"],
        color=colors[group]
    )
    ax.axvline(0, color="black", linewidth=0.8)
    ax.set_xlabel("log10(immunoassay / MS)")
    ax.set_title(f"{group}: ten largest ratios")
    ax.grid(axis="y", visible=False)

plt.tight_layout()
plt.show()
```

The scatter uses **log10 MS concentration on the x-axis** and **log10 immunoassay concentration on the y-axis**; points are coloured orange for liver-enriched proteins and blue for the rest.
