## Scatter plot

The requested scatter has:

Because the tool output is tabular rather than graphical, the following SQL is the exact table-generation query behind the figure. It returns the full gene-level plotting table, including the cohort label.

```sql
WITH pa AS (
    SELECT
        Gene,
        Ensembl AS ensembl_id,
        "Protein class" AS protein_class,
        "RNA tissue specificity" AS tissue_specificity,
        lower(coalesce("RNA tissue specific nTPM", '')) AS tissue_specific_ntpm
    FROM proteinatlas
),

cohorts AS (
    SELECT
        Gene,
        ensembl_id,
        CASE
            WHEN protein_class ILIKE '%Enzymes%'
                 AND tissue_specificity = 'Tissue enhanced'
                 AND tissue_specific_ntpm LIKE '%pancreas:%'
                THEN 'Pancreatic enhanced enzyme'
            WHEN protein_class ILIKE '%Transporters%'
                 AND tissue_specificity = 'Tissue enriched'
                 AND tissue_specific_ntpm LIKE '%kidney:%'
                THEN 'Kidney-enriched transporter'
            ELSE 'Other'
        END AS cohort
    FROM pa
),

tissue_values AS (
    SELECT
        c.Gene,
        c.ensembl_id,
        c.cohort,
        AVG(TRY_CAST(r.nTPM AS DOUBLE))
            FILTER (WHERE lower(r.Tissue) = 'pancreas')
            AS pancreas_ntpm,
        AVG(TRY_CAST(r.nTPM AS DOUBLE))
            FILTER (WHERE lower(r.Tissue) = 'kidney')
            AS kidney_ntpm
    FROM cohorts c
    JOIN rna_tissue_consensus r
        ON r.Gene = c.ensembl_id
    GROUP BY
        c.Gene,
        c.ensembl_id,
        c.cohort
)

SELECT
    Gene,
    ensembl_id,
    pancreas_ntpm,
    kidney_ntpm,
    cohort
FROM tissue_values
WHERE pancreas_ntpm IS NOT NULL
  AND kidney_ntpm IS NOT NULL
ORDER BY cohort, Gene;
```
