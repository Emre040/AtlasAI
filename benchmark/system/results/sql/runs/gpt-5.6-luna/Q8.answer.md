## Reproducible plotting code

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# `df` should contain the HPA paired data returned from:
# proteinatlas:
#   Gene
#   Secretome location
#   Blood concentration - Conc. blood IM [pg/L]
#   Blood concentration - Conc. blood MS [pg/L]

df = df.rename(columns={
    "Gene": "gene",
    "Secretome location": "secretome_location",
    "Blood concentration - Conc. blood IM [pg/L]": "im_pg_L",
    "Blood concentration - Conc. blood MS [pg/L]": "ms_pg_L",
})

df["im_pg_L"] = pd.to_numeric(df["im_pg_L"], errors="coerce")
df["ms_pg_L"] = pd.to_numeric(df["ms_pg_L"], errors="coerce")

# Keep proteins with positive measurements by both methods
df = df[
    df["im_pg_L"].gt(0) &
    df["ms_pg_L"].gt(0)
].copy()

df["log10_im"] = np.log10(df["im_pg_L"])
df["log10_ms"] = np.log10(df["ms_pg_L"])
df["log10_ratio"] = df["log10_im"] - df["log10_ms"]

# 1. Scatter plot: immunoassay against mass spectrometry,
#    coloured by secretome location
plt.figure(figsize=(8, 7))
sns.scatterplot(
    data=df,
    x="log10_ms",
    y="log10_im",
    hue="secretome_location",
    s=35,
    alpha=0.75
)

lims = [
    min(df["log10_ms"].min(), df["log10_im"].min()),
    max(df["log10_ms"].max(), df["log10_im"].max())
]
plt.plot(lims, lims, "--", color="black", linewidth=1)
plt.xlabel("log10 mass-spectrometry concentration [pg/L]")
plt.ylabel("log10 immunoassay concentration [pg/L]")
plt.title("Blood protein concentrations: immunoassay vs mass spectrometry")
plt.tight_layout()
plt.show()

# 2. Diverging bar chart for the fifteen most discordant proteins
top15 = (
    df.assign(abs_log10_ratio=df["log10_ratio"].abs())
      .sort_values("abs_log10_ratio", ascending=False)
      .head(15)
      .sort_values("log10_ratio")
)

plt.figure(figsize=(9, 7))
colors = np.where(top15["log10_ratio"] >= 0, "#C44E52", "#4C72B0")
plt.barh(top15["gene"], top15["log10_ratio"], color=colors)
plt.axvline(0, color="black", linewidth=1)
plt.xlabel("log10(immunoassay / mass spectrometry)")
plt.ylabel("Gene")
plt.title("Fifteen most discordant proteins")
plt.tight_layout()
plt.show()

# 3. Bar chart of proteins by secretome location
location_counts = (
    df["secretome_location"]
      .fillna("(missing)")
      .value_counts()
      .sort_values(ascending=False)
)

plt.figure(figsize=(8, 5))
location_counts.plot(kind="bar", color="#8172B2")
plt.ylabel("Number of proteins")
plt.xlabel("Secretome location")
plt.title("Paired blood-concentration proteins by secretome location")
plt.xticks(rotation=45, ha="right")
plt.tight_layout()
plt.show()
```

Positive log-ratio bars indicate higher immunoassay concentrations, while negative bars indicate higher mass-spectrometry concentrations.
