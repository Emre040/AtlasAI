This schematic highlights the numerical range and the largest-discrepancy genes; the exact full scatter plotting data are reproducibly defined by the following HPA/DuckDB query (one row per plotted point, ordered by Ensembl ID):

```sql
WITH universe AS (
  SELECT Ensembl AS ensembl_id, Gene AS symbol
  FROM proteinatlas
  WHERE lower("Protein class") LIKE '%fda approved drug targets%'
),
rna AS (
  SELECT
    Gene AS ensembl_id,
    max(CASE WHEN Tissue = 'kidney' THEN TRY_CAST(nTPM AS DOUBLE) END) AS kidney_rna_ntpm,
    max(CASE WHEN Tissue = 'colon'  THEN TRY_CAST(nTPM AS DOUBLE) END) AS colon_rna_ntpm
  FROM rna_tissue_consensus
  WHERE Tissue IN ('kidney', 'colon')
  GROUP BY Gene
),
protein AS (
  SELECT
    Gene AS ensembl_id,
    max(CASE WHEN Tissue = 'kidney' THEN TRY_CAST(Intensity AS DOUBLE) END) AS kidney_ms_intensity,
    max(CASE WHEN Tissue = 'colon'  THEN TRY_CAST(Intensity AS DOUBLE) END) AS colon_ms_intensity
  FROM ms_tissue
  WHERE Tissue IN ('kidney', 'colon')
  GROUP BY Gene
)
SELECT
  u.ensembl_id,
  u.symbol,
  log2(r.kidney_rna_ntpm / r.colon_rna_ntpm) AS rna_log2_kidney_over_colon,
  log2(p.kidney_ms_intensity / p.colon_ms_intensity) AS protein_log2_kidney_over_colon
FROM universe AS u
JOIN rna AS r USING (ensembl_id)
JOIN protein AS p USING (ensembl_id)
WHERE r.kidney_rna_ntpm > 0
  AND r.colon_rna_ntpm > 0
  AND p.kidney_ms_intensity > 0
  AND p.colon_ms_intensity > 0
ORDER BY u.ensembl_id;
```
